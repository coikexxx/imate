## Global Constraints

- 界面文案、报错、注释、提交信息使用简体中文。技术术语保留英文。
- 前后端 JSON 一律 camelCase。
- 适配器的解析代码必须基于录制的真实样本来写和测。样本没有覆盖的字段不凭印象解析；确实只有文字记录的失败路径，测试里注明「按实测记录构造，不是录制样本」。
- 提示词不走命令行参数：dsh 经 stdin，grok 经临时文件。临时文件由 runner 写入并在回合结束后删除，权限 0600（Unix）。
- 工作目录只通过子进程的 `current_dir` 设置，不传 `--cwd`。
- 有权限回显的 CLI（grok）必须核对，回显不符立即失败。没有回显的 CLI（dsh）不伪造「已核对」。
- 用量只填入语义有样本依据的字段。grok 的 `result.usage` 与 `modelUsage.*.costUSD` 按本回合计（依据见 Task 3）；dsh 按 `step_end.usage` 逐步相加，不把 `final` 或其他字段再加一遍。Task 7 的实测若推翻这些语义，对应归一字段改回 `None` 并保留 `rawUsage`。
- dsh 没有 `--model`。2026-11-11 用户授权增加认证方式与已配置自定义模型选择：Task 12 通过临时原生插件读取 Dsh 模型目录，并在内存里应用所选 profile 的模型配置；`agents.model` 保存 profile/provider/model，不拼接命令参数。
- Windows 下不经过 `cmd.exe` 启动任何 CLI。`.cmd` 启动器只接受已核对的格式，解析成真实程序和参数后直接启动。
- 讨论模式不是文件系统隔离。grok 的 `plan` 会写自己的会话目录，提示词里的「本回合不得修改文件」保留。
- 不做 M4 的文件树与像素办公室、M6 的通用设置页，不做用户自定义 CLI。2026-11-11 用户新增授权本机代理并扩展为应用统一网络代理，范围见 Task 10/11。

```rust
/// 提示词文件随作用域删除，任何提前返回都不会把它留在磁盘上。
struct PromptFile(Option<PathBuf>);
impl Drop for PromptFile {
    fn drop(&mut self) {
        if let Some(path) = self.0.take() { let _ = std::fs::remove_file(path); }
    }
}

/// 只新建，不覆盖已有文件。Unix 下只有当前用户可读写。
fn write_prompt_file(path: &std::path::Path, prompt: &str) -> std::io::Result<()> {
    use std::io::Write;
    let mut options = std::fs::OpenOptions::new();
    options.write(true).create_new(true);
    #[cfg(unix)]
    { use std::os::unix::fs::OpenOptionsExt; options.mode(0o600); }
    let mut file = options.open(path)?;
    file.write_all(prompt.as_bytes())?;
    file.flush()
}
```

```rust
/// 直接调用 node 的启动器：某一行正好是 `"<node.exe 绝对路径>" "<入口 .js 绝对路径>" %*`。
/// 2026-11-11 在 Windows 的 dsh 0.2.2-rc.1 上看到这种格式。两个文件都必须存在。
fn direct_node_shim(shim: &str) -> Option<(PathBuf, PathBuf)> {
    shim.lines().map(str::trim).find_map(|line| {
        let rest = line.strip_prefix('"')?;
        let (node, rest) = rest.split_once('"')?;
        let rest = rest.trim_start().strip_prefix('"')?;
        let (script, tail) = rest.split_once('"')?;
        if tail.trim() != "%*" { return None; }
        let (node, script) = (PathBuf::from(node), PathBuf::from(script));
        let node_ok = node.is_absolute() && node.is_file()
            && node.file_name().is_some_and(|name| name.eq_ignore_ascii_case("node.exe"));
        let script_ok = script.is_absolute() && script.is_file()
            && script.extension().is_some_and(|ext| ext.eq_ignore_ascii_case("js"));
        (node_ok && script_ok).then_some((node, script))
    })
}
```

```rust
/// 取版本号时实际启动的程序。`.cmd` 启动器先解析成真实入口，不经过 cmd.exe；解析不了就不取版本。
fn version_command(program: &Path, path_env: &OsString) -> Option<(PathBuf, Vec<String>)> {
    crate::runner::platform::resolve_program(program, &["--version".to_string()], Some(path_env.as_os_str())).ok()
}

fn run_version(program: &Path, path_env: &OsString) -> Option<String> {
    let (program, args) = version_command(program, path_env)?;
    let mut command = Command::new(program);
    command.args(args).env("PATH", path_env);
    parse_version(&probe_output(command, Duration::from_secs(3))?)
}
```

```rust
use std::collections::HashSet;
use std::path::PathBuf;
use serde_json::Value;
use super::{AgentEvent, CliAdapter, CommandSpec, RunMode, TurnRequest, Usage};
use crate::cli_registry::{self, CliInfo, ResolvedCli};

/// grok 的无头适配器。事件格式依据 Windows 1.0.77 的 `streaming-messages-json` 样本。
pub struct GrokAdapter {
    program: PathBuf,
    path_env: Option<std::ffi::OsString>,
    mode: RunMode,
    active_tools: HashSet<String>,
    failed: bool,
    completed: bool,
}

impl GrokAdapter {
    pub fn new(program: PathBuf, mode: RunMode) -> Self {
        Self { program, path_env: None, mode, active_tools: HashSet::new(), failed: false, completed: false }
    }
    pub fn with_environment(mut self, resolved: &ResolvedCli) -> Self {
        self.program = resolved.program.clone();
        self.path_env = Some(resolved.path_env.clone());
        self
    }
    /// plan 只是讨论意图：它拦住工作目录里的写，但 grok 仍会写自己的会话目录。
    fn permission(&self) -> &'static str {
        match self.mode { RunMode::Discuss => "plan", RunMode::Execute => "bypassPermissions" }
    }
    fn fail(&mut self, message: String) -> Vec<AgentEvent> {
        self.failed = true;
        vec![AgentEvent::Error { message }]
    }
}

impl CliAdapter for GrokAdapter {
    fn id(&self) -> &'static str { "grok" }
    fn detect(&self) -> Option<CliInfo> { cli_registry::resolve_cli(self.id()).ok().map(|cli| cli.info) }
    fn build_command(&self, req: &TurnRequest) -> CommandSpec {
        // 只决定路径。runner 在启动前写入提示词，回合结束后删除。
        let prompt_file = std::env::temp_dir().join(format!("imates-prompt-{}.md", uuid::Uuid::new_v4()));
        let mut args = vec!["--prompt-file".into(), prompt_file.to_string_lossy().into_owned(),
            "--output-format".into(), "streaming-messages-json".into(),
            "--permission-mode".into(), self.permission().into()];
        if let Some(id) = &req.session_id { args.extend(["-r".into(), id.clone()]); }
        if let Some(model) = req.model.as_ref().filter(|model| !model.trim().is_empty()) {
            args.extend(["-m".into(), model.clone()]);
        }
        CommandSpec { program: self.program.clone(), args, path_env: self.path_env.clone(),
            stdin: None, prompt_file: Some(prompt_file) }
    }
    fn parse_line(&mut self, line: &str) -> Vec<AgentEvent> {
        let Ok(v) = serde_json::from_str::<Value>(line) else { return vec![]; };
        if self.failed || self.completed { return vec![]; }
        match v["type"].as_str() {
            Some("system") if v["subtype"] == "init" => {
                if v["permissionMode"].as_str() != Some(self.permission()) {
                    return self.fail(format!("Grok 权限模式不符：期望 {}，收到 {}", self.permission(),
                        v["permissionMode"].as_str().unwrap_or("未回显")));
                }
                v["session_id"].as_str().map(|id| vec![AgentEvent::SessionStarted { session_id: id.into() }]).unwrap_or_default()
            }
            Some("assistant" | "user") => {
                let mut events = vec![];
                let Some(content) = v["message"]["content"].as_array() else { return events; };
                for block in content {
                    match block["type"].as_str() {
                        Some("text") if v["type"] == "assistant" => {
                            if let Some(text) = block["text"].as_str() { events.push(AgentEvent::TextDelta { text: text.into() }); }
                        }
                        Some("tool_use") => {
                            if let Some(id) = block["id"].as_str() {
                                if self.active_tools.insert(id.into()) {
                                    // 只报工具名，不带参数，避免路径和内容进入界面与日志。
                                    events.push(AgentEvent::ToolUse { name: block["name"].as_str().unwrap_or("工具").into(), summary: "正在运行工具".into() });
                                    events.push(AgentEvent::ToolState { active: true });
                                }
                            }
                        }
                        Some("tool_result") => {
                            if let Some(id) = block["tool_use_id"].as_str() {
                                self.active_tools.remove(id);
                                events.push(AgentEvent::ToolState { active: !self.active_tools.is_empty() });
                            }
                        }
                        _ => {}
                    }
                }
                events
            }
            Some("result") => {
                if v["is_error"].as_bool().unwrap_or(false) || v["subtype"] != "success" {
                    let message = v["errors"].as_array()
                        .map(|errors| errors.iter().filter_map(Value::as_str).collect::<Vec<_>>().join("；"))
                        .filter(|text| !text.is_empty())
                        .or_else(|| v["result"].as_str().filter(|text| !text.is_empty()).map(str::to_owned))
                        .unwrap_or_else(|| "Grok 回合失败".into());
                    return self.fail(message);
                }
                let usage = v.get("usage").filter(|u| u.is_object()).map(|u| {
                    // 样本里每个模型一项 costUSD；任何一项缺失或非法就不给费用。
                    let cost_usd = v["modelUsage"].as_object().filter(|models| !models.is_empty()).and_then(|models| {
                        models.values().map(|m| m["costUSD"].as_f64().filter(|n| n.is_finite() && *n >= 0.0))
                            .collect::<Option<Vec<_>>>().map(|costs| costs.into_iter().sum())
                    });
                    Usage { input_tokens: u["input_tokens"].as_u64(), output_tokens: u["output_tokens"].as_u64(),
                        cache_read_input_tokens: u["cache_read_input_tokens"].as_u64(),
                        cache_creation_input_tokens: u["cache_creation_input_tokens"].as_u64(), cost_usd, raw_usage: Some(u.clone()) }
                });
                self.completed = true;
                vec![AgentEvent::TurnCompleted { final_text: v["result"].as_str().unwrap_or_default().into(), usage }]
            }
            _ => vec![],
        }
    }
}
```

```rust
use std::collections::HashSet;
use std::path::PathBuf;
use serde_json::{json, Value};
use super::{AgentEvent, CliAdapter, CommandSpec, RunMode, TurnRequest, Usage};
use crate::cli_registry::{self, CliInfo, ResolvedCli};

/// dsh（DeepSeek Harness）的无头适配器。事件格式依据 Windows 0.2.2-rc.1 的 `--json` 样本。
/// dsh 不回显权限模式，也没有 `--model`。
pub struct DshAdapter {
    program: PathBuf,
    path_env: Option<std::ffi::OsString>,
    mode: RunMode,
    active_tools: HashSet<String>,
    steps: Vec<Value>,
    had_text: bool,
    failed: bool,
    completed: bool,
}

/// 回合没有正常完成时给用户看的说明。字段名来自实测记录。
fn reason_message(reason: &Value) -> String {
    let error = &reason["error"];
    match (error["code"].as_str(), error["message"].as_str()) {
        (Some(code), Some(message)) => format!("dsh 回合失败（{code}）：{message}"),
        (None, Some(message)) => format!("dsh 回合失败：{message}"),
        (Some(code), None) => format!("dsh 回合失败（{code}）"),
        (None, None) => format!("dsh 回合没有完成（{}）", reason["kind"].as_str().unwrap_or("未知原因")),
    }
}

impl DshAdapter {
    pub fn new(program: PathBuf, mode: RunMode) -> Self {
        Self { program, path_env: None, mode, active_tools: HashSet::new(), steps: Vec::new(),
            had_text: false, failed: false, completed: false }
    }
    pub fn with_environment(mut self, resolved: &ResolvedCli) -> Self {
        self.program = resolved.program.clone();
        self.path_env = Some(resolved.path_env.clone());
        self
    }
    fn fail(&mut self, message: String) -> Vec<AgentEvent> {
        self.failed = true;
        vec![AgentEvent::Error { message }]
    }
    /// 逐步相加 `step_end.usage`。某个字段只要有一步缺失就整项留空，不拿零去补。
    fn usage(&self) -> Option<Usage> {
        if self.steps.is_empty() { return None; }
        let sum = |key: &str| self.steps.iter().map(|step| step[key].as_u64()).collect::<Option<Vec<_>>>().map(|values| values.into_iter().sum());
        Some(Usage { input_tokens: sum("inputTokens"), output_tokens: sum("outputTokens"),
            cache_read_input_tokens: sum("cacheReadTokens"), cache_creation_input_tokens: sum("cacheWriteTokens"),
            cost_usd: None, raw_usage: Some(json!({ "steps": self.steps })) })
    }
}

impl CliAdapter for DshAdapter {
    fn id(&self) -> &'static str { "dsh" }
    fn detect(&self) -> Option<CliInfo> { cli_registry::resolve_cli(self.id()).ok().map(|cli| cli.info) }
    fn build_command(&self, req: &TurnRequest) -> CommandSpec {
        let permission = match self.mode { RunMode::Discuss => "read-only", RunMode::Execute => "workspace-write" };
        let mut args: Vec<String> = vec!["headless".into(), "--json".into()];
        if let Some(id) = &req.session_id { args.extend(["--session-id".into(), id.clone()]); }
        // 模型写在 dsh 自己的 profile 里，这里有意不看 req.model。
        args.extend(["--permission-mode".into(), permission.into(), "--approval-policy".into(), "never".into(), "-".into()]);
        CommandSpec { program: self.program.clone(), args, path_env: self.path_env.clone(),
            stdin: Some(req.prompt.clone()), prompt_file: None }
    }
    fn parse_line(&mut self, line: &str) -> Vec<AgentEvent> {
        let Ok(v) = serde_json::from_str::<Value>(line) else { return vec![]; };
        if self.failed || self.completed { return vec![]; }
        match v["type"].as_str() {
            Some("session") => v["sessionId"].as_str()
                .map(|id| vec![AgentEvent::SessionStarted { session_id: id.into() }]).unwrap_or_default(),
            Some("text") => {
                let Some(text) = v["text"].as_str().filter(|text| !text.is_empty()) else { return vec![]; };
                let text = if self.had_text { format!("\n\n{text}") } else { text.to_string() };
                self.had_text = true;
                vec![AgentEvent::TextDelta { text }]
            }
            Some("tool_call") => {
                let Some(id) = v["callId"].as_str() else { return vec![]; };
                if !self.active_tools.insert(id.into()) { return vec![]; }
                // 只报工具名，不带参数。
                vec![AgentEvent::ToolUse { name: v["tool"].as_str().unwrap_or("工具").into(), summary: "正在运行工具".into() },
                    AgentEvent::ToolState { active: true }]
            }
            Some("tool_result") => {
                let Some(id) = v["callId"].as_str() else { return vec![]; };
                self.active_tools.remove(id);
                vec![AgentEvent::ToolState { active: !self.active_tools.is_empty() }]
            }
            Some("status") => match v["phase"].as_str() {
                Some("step_end") => {
                    if let Some(usage) = v.get("usage").filter(|usage| usage.is_object()) { self.steps.push(usage.clone()); }
                    vec![]
                }
                Some("turn_end") if v["reason"]["kind"] != "completed" => self.fail(reason_message(&v["reason"])),
                _ => vec![],
            },
            Some("final") => {
                if v["reason"]["kind"] != "completed" { return self.fail(reason_message(&v["reason"])); }
                self.completed = true;
                vec![AgentEvent::TurnCompleted { final_text: v["text"].as_str().unwrap_or_default().into(), usage: self.usage() }]
            }
            Some("error") => self.fail(v["message"].as_str().unwrap_or("dsh 返回错误").into()),
            _ => vec![],
        }
    }
}
```
