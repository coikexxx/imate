use std::path::{Component, Path, PathBuf};

use serde::Serialize;

use crate::error::{AppError, AppResult};

/// 文件树和文件监听都不展示的目录名。
pub const IGNORED_DIRS: [&str; 8] = [".git", "node_modules", ".imates", "target", "dist", ".venv", "__pycache__", ".idea"];
/// 单次列出的条目上限，避免超大目录卡住界面。
pub const MAX_ENTRIES: usize = 2000;

#[derive(Debug, Clone, PartialEq, Serialize)]
#[serde(rename_all = "camelCase")]
pub struct FileEntry {
    pub path: String,
    pub name: String,
    pub is_dir: bool,
}

#[derive(Debug, Clone, PartialEq, Serialize)]
#[serde(rename_all = "camelCase")]
pub struct FileListing {
    pub entries: Vec<FileEntry>,
    pub truncated: bool,
}

pub fn is_ignored(name: &str) -> bool {
    IGNORED_DIRS.contains(&name)
}

/// 相对项目文件夹的路径，统一用 `/` 分隔。不在文件夹内或就是文件夹本身时返回 None。
pub fn relative_key(folder: &Path, path: &Path) -> Option<String> {
    let rest = path.strip_prefix(folder).ok()?;
    let parts = rest
        .components()
        .map(|component| match component {
            Component::Normal(part) => Some(part.to_string_lossy().into_owned()),
            _ => None,
        })
        .collect::<Option<Vec<_>>>()?;
    if parts.is_empty() { None } else { Some(parts.join("/")) }
}

pub fn list_files(folder: &Path) -> AppResult<FileListing> {
    if !folder.is_dir() {
        return Err(AppError::Invalid("项目文件夹不存在".into()));
    }
    let mut entries = Vec::new();
    let mut truncated = false;
    walk(folder, folder, &mut entries, &mut truncated)?;
    Ok(FileListing { entries, truncated })
}

fn walk(root: &Path, dir: &Path, entries: &mut Vec<FileEntry>, truncated: &mut bool) -> AppResult<()> {
    let mut children = Vec::new();
    for entry in std::fs::read_dir(dir)? {
        let entry = entry?;
        // file_type 不跟随符号链接；链接一律不展示，避免走出项目文件夹或成环。
        let kind = entry.file_type()?;
        if kind.is_symlink() {
            continue;
        }
        let name = entry.file_name().to_string_lossy().into_owned();
        if kind.is_dir() && is_ignored(&name) {
            continue;
        }
        children.push((name, entry.path(), kind.is_dir()));
    }
    children.sort_by(|a, b| {
        b.2.cmp(&a.2)
            .then_with(|| a.0.to_lowercase().cmp(&b.0.to_lowercase()))
            .then_with(|| a.0.cmp(&b.0))
    });
    for (name, path, is_dir) in children {
        if entries.len() >= MAX_ENTRIES {
            *truncated = true;
            return Ok(());
        }
        let Some(key) = relative_key(root, &path) else { continue };
        entries.push(FileEntry { path: key, name, is_dir });
        if is_dir {
            // 读不了的子目录只是不展开，不让整棵树失败。
            let _ = walk(root, &path, entries, truncated);
            if *truncated {
                return Ok(());
            }
        }
    }
    Ok(())
}

/// 把前端传来的相对路径解析成项目文件夹内的真实路径。
pub fn resolve_inside(folder: &Path, relative: &str) -> AppResult<PathBuf> {
    let rel = Path::new(relative);
    let plain = !relative.trim().is_empty() && rel.components().all(|component| matches!(component, Component::Normal(_)));
    if !plain {
        return Err(AppError::Invalid("文件路径不合法".into()));
    }
    let hidden = rel.components().any(|component| matches!(component, Component::Normal(part) if is_ignored(&part.to_string_lossy())));
    if hidden {
        return Err(AppError::Invalid("该路径不在文件列表中".into()));
    }
    let root = folder.canonicalize().map_err(|_| AppError::Invalid("项目文件夹不存在".into()))?;
    let target = root.join(rel).canonicalize().map_err(|_| AppError::NotFound(format!("文件 {relative}")))?;
    if !target.starts_with(&root) {
        return Err(AppError::Invalid("文件不在项目文件夹内".into()));
    }
    Ok(target)
}

/// 交给系统打开时去掉 Windows 的 `\\?\` 前缀，UNC 形式保持原样。
pub fn display_path(path: &Path) -> String {
    let text = path.to_string_lossy().into_owned();
    match text.strip_prefix(r"\\?\") {
        Some(rest) if !rest.starts_with("UNC\\") => rest.to_string(),
        _ => text,
    }
}

use std::collections::{BTreeSet, HashMap};
use std::path::{Path, PathBuf};
use std::sync::{mpsc, Arc, Mutex, MutexGuard};
use std::time::Duration;

use notify::{RecommendedWatcher, RecursiveMode, Watcher};

use crate::error::{AppError, AppResult};

/// 一段安静期之内合并上报。
const DEBOUNCE: Duration = Duration::from_millis(300);
/// 单批最多合并的原始事件数，持续写入时也能及时上报。
const MAX_RAW: usize = 4096;

/// 一次上报：改动的文件（相对路径）和规章目录是否有变化。
#[derive(Debug, Clone, Default, PartialEq)]
pub struct WatchBatch {
    pub paths: Vec<String>,
    pub rules_touched: bool,
}

/// 把原始路径归成相对路径，去重排序，并丢掉忽略目录。监听不判断是谁改的。
pub fn classify(folder: &Path, raw: &[PathBuf]) -> WatchBatch {
    let mut paths = BTreeSet::new();
    let mut rules_touched = false;
    for path in raw {
        let Some(key) = crate::files::relative_key(folder, path) else { continue };
        if key == ".imates/rules" || key.starts_with(".imates/rules/") {
            rules_touched = true;
            continue;
        }
        if key.split('/').any(crate::files::is_ignored) {
            continue;
        }
        paths.insert(key);
    }
    WatchBatch { paths: paths.into_iter().collect(), rules_touched }
}

struct Entry {
    folder: PathBuf,
    _watcher: RecommendedWatcher,
    changed: BTreeSet<String>,
}

/// 每个项目最多一个监听器。高亮集合只存在内存里。
#[derive(Default)]
pub struct WatchHub {
    entries: Arc<Mutex<HashMap<String, Entry>>>,
}

impl WatchHub {
    fn lock(&self) -> AppResult<MutexGuard<'_, HashMap<String, Entry>>> {
        self.entries.lock().map_err(|_| AppError::Invalid("文件监听状态不可用，请重启应用".into()))
    }

    pub fn ensure(&self, project_id: &str, folder: &Path, on_batch: impl Fn(WatchBatch) + Send + 'static) -> AppResult<bool> {
        // 用规范化后的路径监听，事件里的路径才能和它对上（macOS 的 /private、Windows 的 \\?\）。
        let folder = folder.canonicalize().map_err(|_| AppError::Invalid("项目目录不存在或无法读取".into()))?;
        let mut entries = self.lock()?;
        if entries.get(project_id).is_some_and(|entry| entry.folder == folder) {
            return Ok(false);
        }
        let (tx, rx) = mpsc::channel::<PathBuf>();
        let mut watcher = notify::recommended_watcher(move |result: notify::Result<notify::Event>| {
            let Ok(event) = result else { return };
            if matches!(event.kind, notify::EventKind::Access(_)) {
                return;
            }
            for path in event.paths {
                let _ = tx.send(path);
            }
        })
        .map_err(|error| AppError::Invalid(format!("无法监听项目文件夹：{error}")))?;
        watcher
            .watch(&folder, RecursiveMode::Recursive)
            .map_err(|error| AppError::Invalid(format!("无法监听项目文件夹：{error}")))?;
        let shared = self.entries.clone();
        let id = project_id.to_string();
        let root = folder.clone();
        std::thread::Builder::new()
            .name("imates-fs-watch".into())
            .spawn(move || {
                // 监听器被移除后发送端关闭，循环自然结束。
                while let Ok(first) = rx.recv() {
                    let mut raw = vec![first];
                    while raw.len() < MAX_RAW {
                        match rx.recv_timeout(DEBOUNCE) {
                            Ok(next) => raw.push(next),
                            Err(_) => break,
                        }
                    }
                    let batch = classify(&root, &raw);
                    if batch.paths.is_empty() && !batch.rules_touched {
                        continue;
                    }
                    {
                        let Ok(mut entries) = shared.lock() else { break };
                        match entries.get_mut(&id) {
                            Some(entry) if entry.folder == root => entry.changed.extend(batch.paths.iter().cloned()),
                            _ => break,
                        }
                    }
                    on_batch(batch);
                }
            })
            .map_err(|error| AppError::Invalid(format!("无法启动文件监听线程：{error}")))?;
        // 线程要拿到同一把锁才能记录，所以一定晚于这次插入。
        entries.insert(project_id.to_string(), Entry { folder, _watcher: watcher, changed: BTreeSet::new() });
        Ok(true)
    }

    pub fn remove(&self, project_id: &str) {
        if let Ok(mut entries) = self.entries.lock() {
            entries.remove(project_id);
        }
    }

    pub fn changed(&self, project_id: &str) -> Vec<String> {
        self.entries
            .lock()
            .ok()
            .and_then(|entries| entries.get(project_id).map(|entry| entry.changed.iter().cloned().collect()))
            .unwrap_or_default()
    }

    pub fn reset_changed(&self, project_id: &str) {
        if let Ok(mut entries) = self.entries.lock() {
            if let Some(entry) = entries.get_mut(project_id) {
                entry.changed.clear();
            }
        }
    }
}

/// 64 位 FNV-1a。只用来判断规章内容是否变化，不用于安全目的。
pub fn fnv1a(text: &str) -> String {
    let mut hash: u64 = 0xcbf2_9ce4_8422_2325;
    for byte in text.as_bytes() {
        hash ^= u64::from(*byte);
        hash = hash.wrapping_mul(0x0000_0100_0000_01b3);
    }
    format!("{hash:016x}")
}

/// 用有序的文件名和全文算出的规章版本。没有规章时为 `none`。
pub fn fingerprint(rules: &[RuleDoc]) -> String {
    if rules.is_empty() {
        return "none".to_string();
    }
    let mut text = String::new();
    for rule in rules {
        text.push_str(&rule.filename);
        text.push('\0');
        text.push_str(&rule.content);
        text.push('\0');
    }
    fnv1a(&text)
}

/// 文件名到正文摘要。用来比较两次读取之间哪些规章变了。
pub fn digest_map(rules: &[RuleDoc]) -> std::collections::BTreeMap<String, String> {
    rules.iter().map(|rule| (rule.filename.clone(), fnv1a(&rule.content))).collect()
}

/// 应用之外修改规章后给现有会话看的通知。没有变化时返回 None。
/// 监听只能证明文件变了，这里不写是谁改的。
pub fn external_change_notice(previous: &std::collections::BTreeMap<String, String>, current: &[RuleDoc]) -> Option<String> {
    let now = digest_map(current);
    let added: Vec<&RuleDoc> = current.iter().filter(|rule| !previous.contains_key(&rule.filename)).collect();
    let replaced: Vec<&RuleDoc> = current
        .iter()
        .filter(|rule| previous.get(&rule.filename).is_some_and(|old| Some(old) != now.get(&rule.filename)))
        .collect();
    let deleted: Vec<&String> = previous.keys().filter(|name| !now.contains_key(*name)).collect();
    if added.is_empty() && replaced.is_empty() && deleted.is_empty() {
        return None;
    }
    let names = |items: Vec<&str>| if items.is_empty() { "无".to_string() } else { items.join("、") };
    let mut notice = format!(
        "规章版本：{}\n操作：外部修改\n新增：{}\n替换：{}\n删除：{}\n有效文件：{}\n",
        fingerprint(current),
        names(added.iter().map(|rule| rule.filename.as_str()).collect()),
        names(replaced.iter().map(|rule| rule.filename.as_str()).collect()),
        names(deleted.iter().map(|name| name.as_str()).collect()),
        names(current.iter().map(|rule| rule.filename.as_str()).collect()),
    );
    if current.is_empty() {
        notice.push_str("旧规章全部失效，当前暂无规章");
    } else {
        notice.push_str("此前的规章版本已失效。");
    }
    for rule in added.iter().chain(replaced.iter()) {
        notice.push_str(&format!("\n{} 全文：\n{}", rule.filename, rule.content));
    }
    Some(notice)
}

/// 比对磁盘上的规章和上次记录的摘要。有外部变化时写一条系统消息并返回它。
pub fn sync_rules(&self, project_id: &str) -> AppResult<Option<(Message, String)>> {
    let _guard = self.rules_lock()?;
    let folder = {
        let conn = self.db_lock()?;
        projects::get(&conn, project_id)?.folder_path
    };
    // 项目文件夹整个不在时由项目页提示重新选择，不当成规章被清空。
    if !Path::new(&folder).is_dir() {
        return Ok(None);
    }
    let current = rules::list(Path::new(&folder))?;
    let digest = rules::digest_map(&current);
    let conn = self.db_lock()?;
    let Some(previous) = load_rule_digest(&conn, project_id)? else {
        save_rule_digest(&conn, project_id, &digest)?;
        return Ok(None);
    };
    let Some(notice) = rules::external_change_notice(&previous, &current) else {
        return Ok(None);
    };
    let tx = conn.unchecked_transaction()?;
    let message = messages::insert(&tx, project_id, "system", None, "system", &notice)?;
    save_rule_digest(&tx, project_id, &digest)?;
    tx.commit()?;
    Ok(Some((message, rules::fingerprint(&current))))
}

/// 外部规章变化：写通知并告诉前端刷新。没有变化时什么都不发。
fn emit_rule_sync(app: &tauri::AppHandle, state: &AppState, project_id: &str) {
    if let Ok(Some((message, version))) = state.sync_rules(project_id) {
        emit_events(app, vec![
            ("message:new", json!({"projectId": project_id, "runId": Value::Null, "message": message})),

        ]);

}

/// 确保项目文件夹正在被监听。重复调用没有副作用。
pub fn ensure_project_watch(app: &tauri::AppHandle, state: &AppState, project_id: &str) -> AppResult<()> {
    let folder = {
        let conn = state.db_lock()?;
        projects::get(&conn, project_id)?.folder_path
    };
    let handle = app.clone();
    let id = project_id.to_string();
    let created = state.watch.ensure(project_id, Path::new(&folder), move |batch| {
        let state = handle.state::<AppState>();
        if !batch.paths.is_empty() {
            emit_events(&handle, vec![("files:changed", json!({
                "projectId": id, "paths": batch.paths, "changed": state.watch.changed(&id)}))]);
        }
        if batch.rules_touched {
            emit_rule_sync(&handle, state.inner(), &id);
        }
    })?;
    if created {
        // 应用没开着的时候规章也可能被改过，开始监听时先对一次账。
        emit_rule_sync(app, state, project_id);
    }
    Ok(())
}

/// 一条新指令开始：之前的高亮作废。监听启动失败不挡住发送。
fn begin_change_window(app: &tauri::AppHandle, project_id: &str) {
    let state = app.state::<AppState>();
    let _ = ensure_project_watch(app, state.inner(), project_id);
    state.watch.reset_changed(project_id);
    emit_events(app, vec![("files:changed", json!({"projectId": project_id, "paths": [], "changed": []}))]);
}

```ts
import type { FileEntry } from "../types";

export interface TreeRow {
  path: string;
  name: string;
  isDir: boolean;
  depth: number;
  changed: boolean;
  collapsed: boolean;
}

/** 一个相对路径的全部上级目录，由浅到深。 */
export function ancestors(path: string): string[] {
  const parts = path.split("/");
  return parts.slice(0, -1).map((_, index) => parts.slice(0, index + 1).join("/"));
}

/** 把后端给的先序平铺列表变成当前可见的行。改动的文件连同上级目录一起标记。 */
export function visibleRows(entries: FileEntry[], collapsed: ReadonlySet<string>, changed: readonly string[]): TreeRow[] {
  const marked = new Set(changed);
  for (const path of changed) for (const dir of ancestors(path)) marked.add(dir);
  return entries
    .filter((entry) => !ancestors(entry.path).some((dir) => collapsed.has(dir)))
    .map((entry) => ({
      path: entry.path,
      name: entry.name,
      isDir: entry.isDir,
      depth: entry.path.split("/").length - 1,
      changed: marked.has(entry.path),
      collapsed: entry.isDir && collapsed.has(entry.path),
    }));
}
```

import { useEffect, useState } from "react";
import { api } from "../api";
import { visibleRows } from "../lib/fileTree";
import type { FileListing } from "../types";

interface Props { projectId: string; changed: string[]; tick: number }

/** 只读文件树。点击文件用系统默认程序打开，点击目录折叠或展开。 */
export function FileTree({ projectId, changed, tick }: Props) {
  const [listing, setListing] = useState<FileListing | null>(null);
  const [collapsed, setCollapsed] = useState<Set<string>>(new Set());
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    let disposed = false;
    api.listFiles(projectId)
      .then((next) => { if (!disposed) { setListing(next); setError(null); } })
      .catch((failure) => { if (!disposed) setError(String(failure)); });
    return () => { disposed = true; };
  }, [projectId, tick]);

  function toggle(path: string) {
    setCollapsed((current) => {
      const next = new Set(current);
      if (next.has(path)) next.delete(path); else next.add(path);
      return next;
    });
  }
  function openFile(path: string) {
    api.openFile(projectId, path).catch((failure) => setError(String(failure)));
  }

  if (error && !listing) return <p className="p-3 text-sm text-red-400">{error}</p>;
  if (!listing) return <p className="p-3 text-sm text-slate-500">加载中…</p>;
  const rows = visibleRows(listing.entries, collapsed, changed);
  return (
    <div className="p-2 text-sm">
      {error && <p role="alert" className="px-1 pb-2 text-xs text-red-400">{error}</p>}
      {changed.length > 0 && <p className="px-1 pb-2 text-xs text-amber-300">最近一条指令期间有 {changed.length} 个文件发生变化（不区分是谁改的）</p>}
      {rows.length === 0 && <p className="px-1 py-4 text-center text-xs text-slate-500">项目文件夹里还没有可展示的文件。</p>}
      <ul>
        {rows.map((row) => (
          <li key={row.path}>
            <button type="button" title={row.path} style={{ paddingLeft: `${row.depth * 12 + 4}px` }}
              className={`flex w-full items-center gap-1 rounded py-0.5 pr-1 text-left hover:bg-slate-800 ${row.changed ? "text-amber-300" : "text-slate-300"}`}
              onClick={() => (row.isDir ? toggle(row.path) : openFile(row.path))}>
              <span className="w-3 shrink-0 text-xs text-slate-500">{row.isDir ? (row.collapsed ? "▸" : "▾") : ""}</span>
              <span className="truncate">{row.name}{row.isDir ? "/" : ""}</span>
              {row.changed && <span className="ml-auto shrink-0 text-xs" aria-label="已变化">●</span>}
            </button>
          </li>
        ))}
      </ul>
      {listing.truncated && <p className="px-1 pt-2 text-xs text-slate-500">文件较多，只显示前 2000 项。</p>}
    </div>
  );
}

export const SEAT_COLUMNS = 4;
export const SEAT_ROWS = 2;
export const MAX_SEATS = SEAT_COLUMNS * SEAT_ROWS;
/** 一个工位的逻辑边长（像素）。 */
export const CELL = 64;
export const LOGICAL_W = CELL * SEAT_COLUMNS;
export const LOGICAL_H = CELL * SEAT_ROWS;

/** 工位左上角的逻辑坐标。工位号不合法时返回 null。 */
export function seatOrigin(seatIndex: number): { x: number; y: number } | null {
  if (!Number.isInteger(seatIndex) || seatIndex < 0 || seatIndex >= MAX_SEATS) return null;
  return { x: (seatIndex % SEAT_COLUMNS) * CELL, y: Math.floor(seatIndex / SEAT_COLUMNS) * CELL };
}

/** 能放进容器的最大整数倍数，保证像素不被拉糊。 */
export function integerScale(width: number, height: number): number {
  return Math.max(1, Math.floor(Math.min(width / LOGICAL_W, height / LOGICAL_H)) || 1);
}

/** 32 位 FNV-1a，用来从 ID 得到稳定的配色和动作。 */
export function hashOf(text: string): number {
  let hash = 0x811c9dc5;
  for (let index = 0; index < text.length; index += 1) {
    hash ^= text.charCodeAt(index);
    hash = Math.imul(hash, 0x01000193);
  }
  return hash >>> 0;
}

export const SHIRT_COLORS = [
  "#38bdf8", "#f472b6", "#a3e635", "#fbbf24", "#c084fc", "#fb7185", "#34d399", "#f97316",
] as const;

/** 每个成员按 ID 取一个固定配色。 */
export function shirtColor(agentId: string): string {
  return SHIRT_COLORS[hashOf(agentId) % SHIRT_COLORS.length];
}

export type IdleAct = "coffee" | "nap" | "phone";
const IDLE_ACTS: readonly IdleAct[] = ["coffee", "nap", "phone"];
/** 摸鱼动作每 12 秒换一次。同一时间段内结果稳定，便于测试。 */
export function idleAct(agentId: string, nowMs: number): IdleAct {
  const bucket = Math.floor(nowMs / 12_000);
  return IDLE_ACTS[hashOf(`${agentId}:${bucket}`) % IDLE_ACTS.length];
}

import type { AgentStatus } from "../types";

/** 一张像素图：每个字符一个像素。`.` 透明，`C` 是成员配色，其余查调色板。 */
export type Sprite = readonly string[];
/** 相对工位左上角的一层图。 */
export interface Layer { sprite: Sprite; dx: number; dy: number }

export const PALETTE: Record<string, string> = {
  H: "#2b2b3a", // 头发
  S: "#f1c9a5", // 皮肤
  D: "#1e293b", // 深色描边、眼睛、关着的屏幕
  W: "#f8fafc", // 气泡
  K: "#8b5a2b", // 桌面
  L: "#6b4423", // 桌腿与桌沿
  M: "#475569", // 显示器外框
  B: "#38bdf8", // 屏幕
  G: "#bae6fd", // 屏幕亮处
  R: "#ef4444", // 感叹号
  Y: "#facc15", // 杯子
  Z: "#94a3b8", // 热气与 Z
};

interface Pose { mouthOpen?: boolean; eyesClosed?: boolean; handsOut?: boolean }
function person({ mouthOpen = false, eyesClosed = false, handsOut = false }: Pose): Sprite {
  return [
    "...HHHHHH...",
    "..HHHHHHHH..",
    "..HSSSSSSH..",
    eyesClosed ? "..DDSSSSDD.." : "..SDSSSSDS..",
    "..SSSSSSSS..",
    mouthOpen ? "...SDWWDS..." : "...SSDDSS...",
    "....SSSS....",
    "..CCCCCCCC..",
    ".CCCCCCCCCC.",
    ".CCCCCCCCCC.",
    handsOut ? "S.CCCCCCCC.S" : ".SCCCCCCCCS.",
    "..CCCCCCCC..",
  ];
}
const PERSON = person({});
const PERSON_TALK = person({ mouthOpen: true });
const PERSON_NAP = person({ eyesClosed: true });
const PERSON_TYPE = person({ handsOut: true });

function monitor(screen: readonly [string, string, string, string]): Sprite {
  return ["MMMMMMMMMM", ...screen.map((row) => `M${row}M`), "MMMMMMMMMM", "....MM....", "...MMMM..."];
}
const MONITOR_OFF = monitor(["DDDDDDDD", "DDDDDDDD", "DDDDDDDD", "DDDDDDDD"]);
const MONITOR_A = monitor(["BBBBBBBB", "BGGGGBBB", "BBBBBBBB", "BGGBBBBB"]);
const MONITOR_B = monitor(["BBBBBBBB", "BGGGGGGB", "BGGGBBBB", "BBBBBBBB"]);

const DESK: Sprite = [
  "K".repeat(40),
  "K".repeat(40),
  "L".repeat(40),
  ...Array.from({ length: 5 }, () => `LL${".".repeat(36)}LL`),
];

  return [".WWWWWWWW.", "WWWWWWWWWW", second, third, ".WWWWWWWW.", "..WW......", ".W........"];

  ["..Z.", "YYY.", "YYYY", "YYY."],
];
const ZZZ: readonly Sprite[] = [
  ["..ZZZ.", "...Z..", "..ZZZ.", "......", "......"],
  ["......", "...ZZZ", "....Z.", "...ZZZ", "......"],
];
const PHONE: readonly Sprite[] = [
  ["DDD", "DBD", "DBD", "DBD", "DDD"],
  ["DDD", "DGD", "DGD", "DGD", "DDD"],
];

/** 供测试检查矩形与调色板。 */
export const ALL_SPRITES: readonly Sprite[] = [
  PERSON, PERSON_TALK, PERSON_NAP, PERSON_TYPE, MONITOR_OFF, MONITOR_A, MONITOR_B, DESK,
  ...BUBBLE_THINKING, ...BUBBLE_TALKING, ...MARK_ERROR, ...MUG, ...ZZZ, ...PHONE,
];

/** 名字的锚点：水平居中，画在桌子下方。 */
export const NAME_OFFSET = { x: 32, y: 54 } as const;

// 以下坐标都相对工位左上角，工位边长 64。
const AT = {
  person: { dx: 20, dy: 22 },
  monitor: { dx: 40, dy: 26 },
  desk: { dx: 12, dy: 34 },
  bubble: { dx: 22, dy: 12 },
  mark: { dx: 24, dy: 12 },
  zzz: { dx: 32, dy: 14 },
  mug: { dx: 14, dy: 30 },
  phone: { dx: 15, dy: 29 },
} as const;

function pick<T>(frames: readonly T[], tick: number): T {
  return frames[((tick % frames.length) + frames.length) % frames.length];
}
const at = (sprite: Sprite, where: { dx: number; dy: number }): Layer => ({ sprite, dx: where.dx, dy: where.dy });

/** 没有成员的工位：只有桌子和关着的电脑。 */
export function emptySeatLayers(): Layer[] {
  return [at(MONITOR_OFF, AT.monitor), at(DESK, AT.desk)];
}

/** 一个工位在某一帧要画的全部图层，按从后到前的顺序。约每秒 4 个 tick。 */
export function seatLayers(status: AgentStatus, act: IdleAct, tick: number): Layer[] {
  switch (status) {
    case "thinking":
      return [at(PERSON, AT.person), at(MONITOR_A, AT.monitor), at(DESK, AT.desk), at(pick(BUBBLE_THINKING, tick), AT.bubble)];
    case "talking":
      return [at(pick([PERSON, PERSON_TALK], tick), AT.person), at(MONITOR_A, AT.monitor), at(DESK, AT.desk), at(pick(BUBBLE_TALKING, tick), AT.bubble)];
    case "working":
      return [at(pick([PERSON, PERSON_TYPE], tick), AT.person), at(pick([MONITOR_A, MONITOR_B], tick), AT.monitor), at(DESK, AT.desk)];
    case "error":
      return [at(PERSON, AT.person), at(MONITOR_OFF, AT.monitor), at(DESK, AT.desk), at(pick(MARK_ERROR, tick), AT.mark)];
    default:
      if (act === "nap") return [at(PERSON_NAP, AT.person), at(MONITOR_OFF, AT.monitor), at(DESK, AT.desk), at(pick(ZZZ, tick), AT.zzz)];
      if (act === "phone") return [at(PERSON, AT.person), at(MONITOR_OFF, AT.monitor), at(DESK, AT.desk), at(pick(PHONE, tick), AT.phone)];
      return [at(PERSON, AT.person), at(MONITOR_OFF, AT.monitor), at(DESK, AT.desk), at(pick(MUG, tick), AT.mug)];
  }
}

import { useEffect, useRef } from "react";
import type { Agent, AgentStatus } from "../types";
import { CELL, LOGICAL_H, LOGICAL_W, MAX_SEATS, idleAct, integerScale, seatOrigin, shirtColor } from "./layout";
import { NAME_OFFSET, PALETTE, emptySeatLayers, seatLayers, type Layer } from "./sprites";

interface Props { agents: Agent[]; statuses: Record<string, AgentStatus> }

const STATUS_TEXT: Record<AgentStatus, string> = {
  idle: "空闲", thinking: "思考中", talking: "发言中", working: "干活中", error: "出错",
};
/** 每秒约 4 帧。 */
const TICK_MS = 250;

function paint(ctx: CanvasRenderingContext2D, layer: Layer, x: number, y: number, scale: number, shirt: string) {
  layer.sprite.forEach((row, rowIndex) => {
    for (let column = 0; column < row.length; column += 1) {
      const char = row[column];
      if (char === ".") continue;
      ctx.fillStyle = char === "C" ? shirt : PALETTE[char];
      ctx.fillRect((x + layer.dx + column) * scale, (y + layer.dy + rowIndex) * scale, scale, scale);
    }
  });
}

/** 像素办公室：小人固定在自己的工位上，动画只表达状态。 */
export function OfficeCanvas({ agents, statuses }: Props) {
  const wrap = useRef<HTMLDivElement>(null);
  const canvas = useRef<HTMLCanvasElement>(null);
  // 动画循环只建一次，每帧从这里读最新的成员和状态。
  const latest = useRef({ agents, statuses });
  latest.current = { agents, statuses };

  useEffect(() => {
    const host = wrap.current;
    const node = canvas.current;
    if (!host || !node) return;
    let tick = 0;
    let scale = 1;

    const draw = () => {
      const ctx = node.getContext("2d");
      if (!ctx) return;
      ctx.imageSmoothingEnabled = false;
      ctx.fillStyle = "#0f172a";
      ctx.fillRect(0, 0, node.width, node.height);
      const { agents: members, statuses: current } = latest.current;
      const now = Date.now();
      for (let seat = 0; seat < MAX_SEATS; seat += 1) {
        const origin = seatOrigin(seat);
        if (!origin) continue;
        // 地板分隔线
        ctx.fillStyle = "#1e293b";
        ctx.fillRect(origin.x * scale, (origin.y + CELL - 1) * scale, CELL * scale, scale);
        const member = members.find((agent) => agent.seatIndex === seat);
        const layers = member ? seatLayers(current[member.id] ?? "idle", idleAct(member.id, now), tick) : emptySeatLayers();
        const shirt = member ? shirtColor(member.id) : "#475569";
        for (const layer of layers) paint(ctx, layer, origin.x, origin.y, scale, shirt);
        if (member) {
          const name = member.name.length > 6 ? `${member.name.slice(0, 5)}…` : member.name;
          ctx.fillStyle = "#cbd5e1";
          ctx.font = `${Math.max(10, 6 * scale)}px ui-sans-serif, system-ui, sans-serif`;
          ctx.textAlign = "center";
          ctx.textBaseline = "top";
          ctx.fillText(name, (origin.x + NAME_OFFSET.x) * scale, (origin.y + NAME_OFFSET.y) * scale);
        }
      }
    };
    const resize = () => {
      scale = integerScale(host.clientWidth, host.clientHeight);
      node.width = LOGICAL_W * scale;
      node.height = LOGICAL_H * scale;
      draw();
    };
    const observer = new ResizeObserver(resize);
    observer.observe(host);
    resize();
    const timer = window.setInterval(() => { tick += 1; draw(); }, TICK_MS);
    return () => { observer.disconnect(); window.clearInterval(timer); };
  }, []);

  const summary = agents.length === 0
    ? "像素办公室：还没有成员"
    : `像素办公室：${agents.map((agent) => `${agent.name}${STATUS_TEXT[statuses[agent.id] ?? "idle"]}`).join("，")}`;
  return (
    <div ref={wrap} className="flex h-full min-h-0 items-center justify-center overflow-hidden rounded border border-slate-800 bg-slate-950">
      <canvas ref={canvas} role="img" aria-label={summary} style={{ imageRendering: "pixelated" }} />
    </div>
  );
}
